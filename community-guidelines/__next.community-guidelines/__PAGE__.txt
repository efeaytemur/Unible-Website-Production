      window.localStorage.setItem(storageKey, pathLocale);
      return;
    }

    if (window.location.pathname !== "/") {
      window.localStorage.setItem(storageKey, "en");
      return;
    }

    const savedLocale = window.localStorage.getItem(storageKey);
    const browserLocale = (navigator.languages || [navigator.language])
      .map((language) => language.toLowerCase().split("-")[0])
      .find((language) => supported.includes(language));
    const locale = savedLocale && supported.includes(savedLocale) ? savedLocale : browserLocale;

    if (locale && locale !== "en") {
      window.location.replace(
        "/" + locale + "/" + window.location.search + window.location.hash
      );
    }
  } catch {
    // Keep the English fallback when browser storage is unavailable.
