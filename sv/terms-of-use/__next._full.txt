      return;

      window.localStorage.setItem(storageKey, "en");
      return;

    if (locale && locale !== "en") {
      window.location.replace(
        "/" + locale + "/" + window.location.search + window.location.hash
      );
    }
  } catch {
    // Keep the English fallback when browser storage is unavailable.
  }
